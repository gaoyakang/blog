## 代码

行内 `const n = 42` 与语法高亮代码块：

```typescript
type Post = {
  slug: string;
  title: string;
  date: string;
};

export function sortByDate(posts: Post[]): Post[] {
  return [...posts].sort(
    (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime(),
  );
}
```
